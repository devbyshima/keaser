# Owned by shortcuts. The Shortcut settings page, and the Add Expense and
# Wallet cards drawn inside a stand-in of the system card (-KeaserSnippet).
settings | -KeaserSeed single -KeaserTab settings -KeaserSettingsPage shortcut

# The expense cards (-KeaserSnippet): confirm is the interactive card of iOS
# 26, confirm-plain the one of iOS 18 to 25, result the Add Expense card with
# confirmation off, wallet the Wallet automation's.
confirm | -KeaserSeed single -KeaserSnippet confirm
confirm-plain | -KeaserSeed single -KeaserSnippet confirmPlain
result | -KeaserSeed single -KeaserSnippet result
wallet | -KeaserSeed demo -KeaserSnippet wallet
confirm-long | -KeaserSeed single -KeaserSnippet confirm -KeaserSnippetLong 1

# The interactive card with a detail tapped: its options drawn inside the
# card (the chosen one checked, then Go Back). Long lists take two columns,
# then pages that More steps through.
confirm-account | -KeaserSeed demo -KeaserSnippet confirmAccount
confirm-category | -KeaserSeed single -KeaserSnippet confirmCategory
confirm-payment | -KeaserSeed single -KeaserSnippet confirmPayment
confirm-category-more | -KeaserSeed single -KeaserSnippet confirmCategory -KeaserSnippetOptions many
confirm-category-more-2 | -KeaserSeed single -KeaserSnippet confirmCategory -KeaserSnippetOptions many -KeaserSnippetPage 2
confirm-account-long | -KeaserSeed demo -KeaserSnippet confirmAccount -KeaserSnippetLong 1
# At larger text sizes the open list takes fewer lines, so the card stays
# under 340 points (ShortcutCardList.lines): seven up to Large, five at the
# largest standard size (XXXL) and at AX1, four at AX2 and three from AX3 up,
# paged with More. At the accessibility sizes every list runs down one
# column, and the header shows the value in place of its label when both do
# not fit.
confirm-category-more-xxxl | -KeaserSeed single -KeaserSnippet confirmCategory -KeaserSnippetOptions many -UIPreferredContentSizeCategoryName UICTContentSizeCategoryXXXL
confirm-category-ax | -KeaserSeed single -KeaserSnippet confirmCategory -UIPreferredContentSizeCategoryName UICTContentSizeCategoryAccessibilityL
confirm-category-axxxl | -KeaserSeed single -KeaserSnippet confirmCategory -UIPreferredContentSizeCategoryName UICTContentSizeCategoryAccessibilityXXXL

# The closed card at an accessibility size keeps one line per detail, to stay
# within the system's height (only the expense's details stack).
confirm-ax | -KeaserSeed single -KeaserSnippet confirm -UIPreferredContentSizeCategoryName UICTContentSizeCategoryAccessibilityL
